## Step 2: Update UIMaterial Uniform Block (Industry Standard Approach)

### Current UIMaterial (in ui.frag):
```glsl
layout(set = 1, binding = 0) uniform UIMaterial {
    vec4 tint;         // Color tint/multiplier
    float opacity;     // Global opacity
    int textureMode;   // 0=solid, 1=texture, 2=texture+color blend
    int roundedCorners; // Enable rounded corner rendering
    float cornerRadius; // Corner radius in pixels
} uiMaterial;
```

### Expanded UIMaterial (for both UI and SDF):
```glsl
layout(set = 1, binding = 0) uniform UIMaterial {
    // Existing UI properties (keep unchanged)
    vec4 tint;           // Color tint/multiplier (also used as text color for SDF)
    float opacity;       // Global opacity
    int textureMode;     // 0=solid, 1=texture, 2=texture+color blend, 3=SDF
    int roundedCorners;  // Enable rounded corner rendering
    float cornerRadius;  // Corner radius in pixels
    
    // SDF-specific properties (new)
    float sdfThreshold;  // SDF cutoff threshold (default 0.5)
    float outlineWidth;  // Outline thickness (default 0.0 = no outline)
    vec4 outlineColor;   // Outline color (default black)
    float sdfRange;      // Distance field range (default 4.0)
    
    // Future expansion (padding for compatibility)
    float reserved1;
    float reserved2;
} uiMaterial;
```

## Step 3: Update Both Shaders to Use Unified UIMaterial

### Update ui.frag (existing shader):
No changes needed - existing code still works since we only added new fields.

### Update ui_sdf.frag (new shader):
```glsl
#version 450

//Input from vertex shader (same as ui.frag)
layout(location = 0) in vec2 texCoords;
layout(location = 1) in vec4 vertexColor;

//Output
layout(location = 0) out vec4 outColor;

//Unified UI Material (same as ui.frag)
layout(set = 1, binding = 0) uniform UIMaterial {
    vec4 tint;
    float opacity;
    int textureMode;
    int roundedCorners;
    float cornerRadius;
    float sdfThreshold;
    float outlineWidth;
    vec4 outlineColor;
    float sdfRange;
    float reserved1;
    float reserved2;
} uiMaterial;

//SDF texture atlas (same binding as ui.frag)
layout(set = 1, binding = 1) uniform sampler2D uiTexture;

void main() {
    //1. Sample distance field from texture
    float distance = texture(uiTexture, texCoords).r;

    //2. Calculate smoothing based on screen-space derivatives
    float smoothWidth = fwidth(distance);

    //3. Convert distance to alpha using smoothstep
    float alpha = smoothstep(uiMaterial.sdfThreshold - smoothWidth, 
                           uiMaterial.sdfThreshold + smoothWidth, distance);

    //4. Apply text color (reuse tint as text color)
    vec3 textColor = uiMaterial.tint.rgb * vertexColor.rgb;
    float finalAlpha = alpha * uiMaterial.tint.a * vertexColor.a;

    //5. Output final color
    outColor = vec4(textColor, finalAlpha);
}
```

### Material usage examples:
```swift
// Regular UI material
let uiMaterial = UIMaterialUniforms(
    tint: .white,
    opacity: 1.0,
    textureMode: 1, // texture mode
    roundedCorners: 0,
    cornerRadius: 0.0,
    sdfThreshold: 0.5,    // Default (ignored for UI)
    outlineWidth: 0.0,    // Default (ignored for UI)
    outlineColor: .black, // Default (ignored for UI)
    sdfRange: 4.0,        // Default (ignored for UI)
    reserved1: 0.0,
    reserved2: 0.0
)

// SDF text material
let textMaterial = UIMaterialUniforms(
    tint: .red,           // Text color
    opacity: 1.0,
    textureMode: 3,       // SDF mode
    roundedCorners: 0,    // Ignored for text
    cornerRadius: 0.0,    // Ignored for text
    sdfThreshold: 0.5,    // SDF cutoff
    outlineWidth: 0.1,    // Thin outline
    outlineColor: .black, // Black outline
    sdfRange: 4.0,        // SDF range
    reserved1: 0.0,
    reserved2: 0.0
)
```

## Benefits of This Approach

✅ **Industry Proven** - Same as Unreal Engine
✅ **Same Pipeline** - UI and text use identical Vulkan pipeline
✅ **Easy Batching** - Can batch text with other UI elements
✅ **Simple Switching** - Just upload different uniform values
✅ **Future Proof** - Easy to add more material types
✅ **Backward Compatible** - Existing UI code unchanged
